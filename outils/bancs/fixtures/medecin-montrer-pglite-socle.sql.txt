CREATE OR REPLACE FUNCTION public.salle_cle_interne(p_pays text, p_ville text, p_batiment text, p_piece text, p_rue text)
 RETURNS text
 LANGUAGE sql
 IMMUTABLE
AS $function$
  SELECT coalesce(p_pays, '') || '|' || coalesce(p_ville, '') || '|'
      || CASE WHEN coalesce(p_batiment, '') <> '' THEN p_batiment || '|' || coalesce(p_piece, '') ELSE 'rue|' || coalesce(p_rue, '') END;
$function$;

CREATE OR REPLACE FUNCTION public.argent_crediter_interne(p_nom text, p_montant numeric, p_poche text DEFAULT 'nationale'::text)
 RETURNS numeric
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE v_m numeric := round(coalesce(p_montant, 0)); v_id text;
BEGIN
  IF p_poche IS NULL OR p_poche NOT IN ('liquide', 'nationale') THEN RAISE EXCEPTION 'poche_invalide'; END IF;
  IF p_nom IS NULL OR v_m <= 0 THEN RETURN 0; END IF;
  PERFORM 1 FROM public.personnages_donnees WHERE name = p_nom FOR UPDATE;
  IF NOT FOUND THEN RETURN 0; END IF;
  IF p_poche = 'nationale' THEN
    SELECT id INTO v_id FROM public.comptes_bancaires
     WHERE personnage = p_nom AND banque = 'nationale' ORDER BY id LIMIT 1 FOR UPDATE;
    IF v_id IS NOT NULL THEN
      UPDATE public.comptes_bancaires SET solde = coalesce(solde, 0) + v_m, updated_at = now() WHERE id = v_id;
      RETURN v_m;
    END IF;
  END IF;
  UPDATE public.personnages_donnees SET liquide = coalesce(liquide, 0) + v_m, updated_at = now() WHERE name = p_nom;
  RETURN v_m;
END;
$function$;

CREATE OR REPLACE FUNCTION public.argent_debiter_interne(p_nom text, p_montant numeric, p_poches text DEFAULT 'liquide_puis_nationale'::text)
 RETURNS numeric
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE v_m numeric := round(coalesce(p_montant, 0)); v_liq numeric; v_id text; v_solde numeric;
        v_pris_l numeric := 0; v_pris_n numeric := 0;
BEGIN
  IF p_poches IS NULL OR p_poches NOT IN ('liquide', 'liquide_puis_nationale') THEN RAISE EXCEPTION 'poches_invalides'; END IF;
  IF p_nom IS NULL OR v_m <= 0 THEN RETURN 0; END IF;
  SELECT coalesce(liquide, 0) INTO v_liq FROM public.personnages_donnees WHERE name = p_nom FOR UPDATE;
  IF NOT FOUND THEN RETURN 0; END IF;
  v_pris_l := least(greatest(v_liq, 0), v_m);
  IF p_poches = 'liquide_puis_nationale' AND v_pris_l < v_m THEN
    SELECT id, coalesce(solde, 0) INTO v_id, v_solde FROM public.comptes_bancaires
     WHERE personnage = p_nom AND banque = 'nationale' ORDER BY id LIMIT 1 FOR UPDATE;
    IF v_id IS NOT NULL THEN v_pris_n := least(greatest(v_solde, 0), v_m - v_pris_l); END IF;
  END IF;
  IF v_pris_l > 0 THEN
    UPDATE public.personnages_donnees SET liquide = v_liq - v_pris_l, updated_at = now() WHERE name = p_nom;
  END IF;
  IF v_pris_n > 0 THEN
    UPDATE public.comptes_bancaires SET solde = solde - v_pris_n, updated_at = now() WHERE id = v_id;
  END IF;
  RETURN v_pris_l + v_pris_n;
END;
$function$;

CREATE OR REPLACE FUNCTION public.surveillance_lieu_interne(p_nom text, OUT pays text, OUT ville text, OUT batiment text, OUT piece text)
 LANGUAGE sql STABLE AS $$ SELECT country, current_city, coalesce(current_building, 'rue'), coalesce(current_room, current_street) FROM public.personnages_donnees WHERE name = p_nom $$;
CREATE OR REPLACE FUNCTION public.inspecter_piece()
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE a text := public.mon_personnage(); l record; d record;
BEGIN
  IF a IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'acteur_non_authentifie'); END IF;
  SELECT name, pa, country, current_city, current_building, current_room INTO d FROM public.personnages_donnees WHERE name = a FOR UPDATE;
  SELECT * INTO l FROM public.surveillance_lieu_interne(a);
  IF l.batiment IS NULL OR l.piece IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'lieu_invalide'); END IF;
  IF coalesce(d.pa, 0) < 1 THEN RETURN jsonb_build_object('ok', false, 'raison', 'pa_insuffisants'); END IF;
  UPDATE public.personnages_donnees SET pa = pa - 1 WHERE name = a;
  RETURN jsonb_build_object('ok', true, 'pa', d.pa - 1);
END $function$;

CREATE OR REPLACE FUNCTION public.desinstaller_lire()
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE a text := public.mon_personnage(); d record;
BEGIN
  IF a IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'acteur_non_authentifie'); END IF;
  SELECT country, current_city, current_building, current_room, pa INTO d FROM public.personnages_donnees WHERE name = a;
  IF d.current_building IS NULL OR d.current_room IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'hors_batiment'); END IF;
  RETURN jsonb_build_object('ok', true, 'materiel', '[]'::jsonb);
END $function$;

CREATE OR REPLACE FUNCTION public.desinstaller_materiel(p_requete uuid, p_id uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE a text := public.mon_personnage(); m record;
BEGIN
  IF a IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'acteur_non_authentifie'); END IF;
  IF p_id IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'equipement_introuvable'); END IF;
  SELECT name, country, current_city, current_building, current_room, pa INTO m FROM public.personnages_donnees WHERE name = a FOR UPDATE;
  IF m.current_building IS NULL OR m.current_room IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'hors_batiment'); END IF;
  RETURN jsonb_build_object('ok', false, 'raison', 'equipement_introuvable');
END $function$;
