CREATE OR REPLACE FUNCTION public.aliment_recu_interne(p_e jsonb) RETURNS date LANGUAGE sql IMMUTABLE AS $$ SELECT nullif(p_e->>'recu', '')::date $$;

CREATE OR REPLACE FUNCTION public.inventaire_pile_fusionnable(p_e jsonb, p_o jsonb) RETURNS boolean LANGUAGE sql IMMUTABLE AS
$$ SELECT coalesce(p_e->>'stackKey' = p_o->>'stackKey', false) $$;
CREATE OR REPLACE FUNCTION public.inventaire_place_restante(p_inv jsonb) RETURNS integer LANGUAGE sql IMMUTABLE AS $$
 SELECT greatest(0,100-coalesce((SELECT sum(CASE WHEN coalesce(e->>'qty','') ~ '^[0-9]+$' THEN (e->>'qty')::numeric
   WHEN coalesce(e->>'encombrement','') ~ '^[0-9]+$' THEN (e->>'encombrement')::numeric ELSE 1 END)
   FROM jsonb_array_elements(coalesce(p_inv,'[]'::jsonb)) e),0))::integer $$;
CREATE OR REPLACE FUNCTION public.inventaire_deposer_interne(p_nom text, p_objet jsonb, p_automatique boolean DEFAULT false)
RETURNS jsonb LANGUAGE plpgsql AS $$
DECLARE v_inv jsonb; v_place integer; v_q integer; v_cle text; v_obj jsonb := p_objet; v_pile bigint;
BEGIN
  SELECT coalesce(inventory,'[]'::jsonb) INTO v_inv FROM public.personnages_donnees WHERE name = p_nom FOR UPDATE;
  v_place := CASE WHEN p_automatique THEN 1000000 ELSE public.inventaire_place_restante(v_inv) END;
  v_cle := nullif(p_objet ->> 'stackKey', '');
  IF coalesce((p_objet ->> 'stackable')::boolean, false) AND v_cle IS NOT NULL THEN
    v_q := least(coalesce((p_objet->>'qty')::numeric,1),v_place)::integer;
    IF v_q <= 0 THEN RETURN jsonb_build_object('ok', false, 'raison', 'inventaire_plein'); END IF;
    SELECT o INTO v_pile FROM jsonb_array_elements(v_inv) WITH ORDINALITY x(e,o) WHERE coalesce((e->>'stackable')::boolean,false) AND e->>'stackKey'=v_cle ORDER BY o LIMIT 1;
    IF v_pile IS NOT NULL THEN
      SELECT jsonb_agg(CASE WHEN o=v_pile THEN jsonb_set(e, '{qty}', to_jsonb(coalesce((e ->> 'qty')::numeric, 1) + v_q)) ELSE e END ORDER BY o)
        INTO v_inv FROM jsonb_array_elements(v_inv) WITH ORDINALITY x(e, o);
    ELSE v_inv := v_inv || jsonb_build_array(p_objet || jsonb_build_object('qty', v_q)); END IF;
  ELSE
    v_q := greatest(1, coalesce((p_objet ->> 'encombrement')::numeric, 1)::integer);
    IF v_place < v_q THEN RETURN jsonb_build_object('ok', false, 'raison', 'inventaire_plein'); END IF;
    IF coalesce(p_objet ->> 'id', '') = '' THEN v_obj := p_objet || jsonb_build_object('id', 'obj-' || replace(gen_random_uuid()::text, '-', '')); END IF;
    v_inv := v_inv || jsonb_build_array(v_obj); v_q := 1;
  END IF;
  UPDATE public.personnages_donnees SET inventory = v_inv WHERE name = p_nom;
  RETURN jsonb_build_object('ok', true, 'qte', v_q, 'inventory', v_inv);
END $$;
CREATE OR REPLACE FUNCTION public.argent_crediter_interne(p_nom text, p_montant numeric, p_poche text DEFAULT 'nationale') RETURNS numeric LANGUAGE plpgsql AS $$
DECLARE v_m numeric := round(coalesce(p_montant, 0));
BEGIN
  IF v_m <= 0 THEN RETURN 0; END IF;
  UPDATE public.personnages_donnees SET liquide = coalesce(liquide, 0) + v_m WHERE name = p_nom; RETURN v_m;
END $$;
CREATE OR REPLACE FUNCTION public.argent_debiter_interne(p_nom text, p_montant numeric, p_poches text DEFAULT 'liquide_puis_nationale') RETURNS numeric LANGUAGE plpgsql AS $$
DECLARE v_m numeric := round(coalesce(p_montant, 0)); v_liq numeric; v_id text; v_solde numeric; v_pris_l numeric := 0; v_pris_n numeric := 0;
BEGIN
  IF v_m <= 0 THEN RETURN 0; END IF;
  SELECT coalesce(liquide, 0) INTO v_liq FROM public.personnages_donnees WHERE name = p_nom FOR UPDATE;
  v_pris_l := least(greatest(v_liq, 0), v_m);
  IF p_poches = 'liquide_puis_nationale' AND v_pris_l < v_m THEN
    SELECT id, coalesce(solde, 0) INTO v_id, v_solde FROM public.comptes_bancaires WHERE personnage = p_nom AND banque = 'nationale' ORDER BY id LIMIT 1 FOR UPDATE;
    IF v_id IS NOT NULL THEN v_pris_n := least(greatest(v_solde, 0), v_m - v_pris_l); END IF;
  END IF;
  IF v_pris_l > 0 THEN UPDATE public.personnages_donnees SET liquide = v_liq - v_pris_l WHERE name = p_nom; END IF;
  IF v_pris_n > 0 THEN UPDATE public.comptes_bancaires SET solde = solde - v_pris_n WHERE id = v_id; END IF;
  RETURN v_pris_l + v_pris_n;
END $$;
CREATE OR REPLACE FUNCTION public.caisse_institution_mouvement(p_id text, p_delta numeric, p_exiger_existant boolean DEFAULT false) RETURNS jsonb LANGUAGE plpgsql AS $$
DECLARE v_solde numeric;
BEGIN
  SELECT (data->>'solde')::numeric INTO v_solde FROM public.caisses_batiments WHERE id = p_id FOR UPDATE;
  IF coalesce(v_solde, 0) + p_delta < 0 THEN RETURN '{"ok":false,"raison":"solde_insuffisant"}'; END IF;
  UPDATE public.caisses_batiments SET data = jsonb_build_object('solde', coalesce(v_solde, 0) + p_delta) WHERE id = p_id;
  RETURN '{"ok":true}';
END $$;

CREATE TABLE public.repas_differes_nuits (nom text, cle text, nuit text, pris_le timestamptz DEFAULT now(), PRIMARY KEY (nom, cle));
CREATE OR REPLACE FUNCTION public.repas_differe_deja_pris_interne(p_nom text, p_cle text) RETURNS boolean LANGUAGE sql STABLE AS $$
  SELECT EXISTS (SELECT 1 FROM public.repas_differes_nuits r JOIN public.personnages_donnees p ON p.name = r.nom
                  WHERE r.nom = p_nom AND r.cle = p_cle AND r.nuit = coalesce(p.pa_repos_le::text, 'jamais')) $$;
CREATE OR REPLACE FUNCTION public.repas_differe_crediter_interne(p_nom text, p_cle text, p_source text) RETURNS integer LANGUAGE plpgsql AS $$
DECLARE v_b integer;
BEGIN
  INSERT INTO public.repas_differes_nuits (nom, cle, nuit) VALUES (p_nom, p_cle, 'jamais') ON CONFLICT (nom, cle) DO UPDATE SET nuit = 'jamais';
  UPDATE public.personnages_donnees SET bonus_pa_differe = coalesce(bonus_pa_differe, 0) + 1 WHERE name = p_nom RETURNING bonus_pa_differe INTO v_b;
  RETURN v_b;
END $$;
