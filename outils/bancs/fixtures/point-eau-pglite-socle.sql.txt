CREATE OR REPLACE FUNCTION public.payer_ordre_interne(p_acteur text, p_fn text, p_pa integer, p_cost integer) RETURNS jsonb LANGUAGE plpgsql AS $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM public.ordres_couts WHERE fn = p_fn AND pa = p_pa AND cost = p_cost) THEN RETURN '{"ok":false,"raison":"ordre_inconnu"}'; END IF;
  IF (SELECT pa FROM public.personnages_donnees WHERE name = p_acteur) < p_pa THEN RETURN '{"ok":false,"raison":"pa_insuffisants"}'; END IF;
  UPDATE public.personnages_donnees SET pa = pa - p_pa WHERE name = p_acteur;
  RETURN jsonb_build_object('ok', true, 'pa_preleves', p_pa);
END $$;
CREATE OR REPLACE FUNCTION public.inventaire_place_restante(p_inv jsonb) RETURNS integer LANGUAGE sql IMMUTABLE AS $$
 SELECT greatest(0,100-coalesce((SELECT sum(CASE WHEN coalesce(e->>'qty','') ~ '^[0-9]+$' THEN (e->>'qty')::numeric ELSE 1 END)
   FROM jsonb_array_elements(coalesce(p_inv,'[]'::jsonb)) e),0))::integer $$;
CREATE OR REPLACE FUNCTION public.inventaire_deposer_interne(p_nom text, p_objet jsonb, p_automatique boolean DEFAULT false) RETURNS jsonb LANGUAGE plpgsql AS $$
DECLARE v_inv jsonb;
BEGIN
  SELECT inventory INTO v_inv FROM public.personnages_donnees WHERE name = p_nom FOR UPDATE;
  IF public.inventaire_place_restante(v_inv) < (p_objet->>'qty')::integer THEN RETURN '{"ok":false,"raison":"inventaire_plein"}'; END IF;
  UPDATE public.personnages_donnees SET inventory = coalesce(inventory, '[]'::jsonb) || jsonb_build_array(p_objet) WHERE name = p_nom RETURNING inventory INTO v_inv;
  RETURN jsonb_build_object('ok', true, 'inventory', v_inv);
END $$;
CREATE OR REPLACE FUNCTION public.climat_objet_interne(p_cle text, p_qte integer) RETURNS jsonb LANGUAGE sql IMMUTABLE AS $$
  SELECT jsonb_build_object('name', 'Bouteille d''eau', 'type', 'bouteille_eau', 'stackable', true, 'stackKey', 'bouteille_eau', 'qty', p_qte) $$;
CREATE OR REPLACE FUNCTION public.climat_consigne_interne(p_type text, p_pays text, p_ville text)
RETURNS text LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path TO 'public', 'pg_temp' AS $function$
DECLARE v_eau text;
BEGIN
  IF p_type = 'canicule' THEN
    SELECT 'Remplissez-la gratuitement à ' || nom || '.' INTO v_eau FROM public.climat_points_eau
     WHERE actif AND pays = p_pays AND ville = p_ville ORDER BY id LIMIT 1;
    RETURN 'Gardez une bouteille d''eau sur vous. ' || coalesce(v_eau, '');
  END IF;
  RETURN NULL;
END $function$;
