CREATE OR REPLACE FUNCTION public.logements_sociaux_terrain_communal(p_etat jsonb) RETURNS boolean LANGUAGE sql IMMUTABLE AS $function$
  SELECT coalesce(p_etat->>'proprietaire','') LIKE 'ville:%' OR coalesce(p_etat->>'acquisitionAcquereur','') LIKE 'ville:%' $function$;
CREATE OR REPLACE FUNCTION public.terrain_permis_deposer(p_terrain text, p_palier text, p_requete uuid) RETURNS jsonb LANGUAGE plpgsql AS $function$
DECLARE duree integer; etat jsonb := '{}'; doc jsonb;
BEGIN
 duree:=CASE p_palier WHEN 'hangar' THEN 2 WHEN 'commerce_standard' THEN 4 WHEN 'commerce_premium' THEN 6 WHEN 'building' THEN 7 WHEN 'logements_petit' THEN 4 WHEN 'logements_moyen' THEN 6 WHEN 'logements_barre' THEN 7 END;
 IF duree IS NULL THEN RETURN jsonb_build_object('ok',false,'raison','palier_invalide'); END IF;
 etat := coalesce((SELECT data::jsonb FROM public.terrains_etat WHERE id = p_terrain), '{}');
 IF p_palier LIKE 'logements_%' THEN
  IF NOT public.logements_sociaux_terrain_communal(etat) THEN RETURN jsonb_build_object('ok',false,'raison','logements_reserves_commune');END IF;
  IF (p_palier='logements_moyen' AND coalesce((etat->>'surface')::numeric,0)<2000) OR (p_palier='logements_barre' AND coalesce((etat->>'surface')::numeric,0)<=2200) THEN RETURN jsonb_build_object('ok',false,'raison','terrain_trop_petit');END IF;
 END IF;
 doc:=jsonb_build_object('palierLabel',CASE p_palier WHEN 'hangar' THEN 'Hangar'
   WHEN 'commerce_standard' THEN 'Commerce standard' WHEN 'commerce_premium' THEN 'Commerce premium' WHEN 'logements_petit' THEN 'Logements sociaux — petit bâtiment' WHEN 'logements_moyen' THEN 'Logements sociaux — bâtiment moyen' WHEN 'logements_barre' THEN 'Logements sociaux — barre d''immeuble' ELSE 'Building' END,
  'duree',duree);
 RETURN jsonb_build_object('ok',true,'document',doc);
END $function$;
CREATE OR REPLACE FUNCTION public.terrain_permis_decider(p_terrain text, p_ok boolean, p_motif text, p_requete uuid) RETURNS jsonb LANGUAGE plpgsql AS $function$
DECLARE lib text; pm jsonb := jsonb_build_object('palierDemande', p_terrain);
BEGIN
  lib := CASE pm->>'palierDemande' WHEN 'hangar' THEN 'Hangar' WHEN 'commerce_standard' THEN 'Commerce standard'
           WHEN 'commerce_premium' THEN 'Commerce premium' WHEN 'building' THEN 'Building' WHEN 'logements_petit' THEN 'Logements sociaux — petit bâtiment' WHEN 'logements_moyen' THEN 'Logements sociaux — bâtiment moyen' WHEN 'logements_barre' THEN 'Logements sociaux — barre d''immeuble' ELSE coalesce(pm->>'palierDemande','') END;
  RETURN jsonb_build_object('lib', lib);
END $function$;
CREATE OR REPLACE FUNCTION public.terrain_acquisition_commande(p_terrain text, p_action text, p_parametres jsonb, p_requete uuid) RETURNS jsonb LANGUAGE plpgsql AS $function$
DECLARE permis text := p_parametres->>'permis';
BEGIN
  IF permis IS NOT NULL AND (p_action<>'compromis' OR permis NOT IN('hangar','commerce_standard','commerce_premium','building','logements_petit','logements_moyen','logements_barre')) THEN RETURN jsonb_build_object('ok',false,'raison','palier_invalide');END IF;
  RETURN jsonb_build_object('ok',true);
END $function$;
