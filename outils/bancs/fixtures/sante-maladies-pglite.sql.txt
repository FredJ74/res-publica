CREATE OR REPLACE FUNCTION public.maladies_echeances_interne(p_nom text)
 RETURNS integer
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE v_n integer;
BEGIN
  UPDATE public.personnages_maladies SET guerie_le = now(), motif_fin = 'echeance'
   WHERE nom = p_nom AND guerie_le IS NULL AND fin <= now();
  GET DIAGNOSTICS v_n = ROW_COUNT;
  IF v_n > 0 THEN PERFORM public.maladie_effets_synchroniser_interne(p_nom); END IF;
  RETURN v_n;
END $function$;

CREATE OR REPLACE FUNCTION public.maladie_effets_synchroniser_interne(p_nom text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE v_now numeric := extract(epoch FROM clock_timestamp()) * 1000; v_base jsonb; v_eff jsonb;
BEGIN
  SELECT CASE WHEN jsonb_typeof(effets_actifs) = 'array' THEN effets_actifs ELSE '[]'::jsonb END
    INTO v_base FROM public.personnages_donnees WHERE name = p_nom;
  IF NOT FOUND THEN RETURN '[]'::jsonb; END IF;
  SELECT coalesce(jsonb_agg(e), '[]'::jsonb) INTO v_eff FROM jsonb_array_elements(v_base) e
   WHERE jsonb_typeof(e) = 'object' AND coalesce(e->>'cle', '') NOT LIKE 'maladie\_%'
     AND coalesce((e->>'expireA')::numeric, 0) > v_now;
  SELECT v_eff || coalesce(jsonb_agg(jsonb_build_object(
           'cle', 'maladie_' || pm.cle, 'cible', 'pj', 'deltas', mp.deltas,
           'capacite', CASE WHEN mp.sans_bonus_arme THEN 'sans_bonus_arme' END,
           'paMax', mp.pa_max, 'maladie', pm.cle,
           'libelle', coalesce(mp.nom_mj, pm.cle), 'poseA', extract(epoch FROM pm.debut) * 1000,
           'expireA', extract(epoch FROM pm.fin) * 1000)), '[]'::jsonb)
    INTO v_eff
    FROM public.personnages_maladies pm JOIN public.maladies_parametres mp ON mp.cle = pm.cle
   WHERE pm.nom = p_nom AND pm.guerie_le IS NULL AND pm.fin > now();
  UPDATE public.personnages_donnees SET effets_actifs = v_eff WHERE name = p_nom;
  RETURN v_eff;
END $function$;

CREATE OR REPLACE FUNCTION public.maladie_contracter_interne(p_nom text, p_cle text, p_source text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE m public.maladies_parametres; v_id bigint;
BEGIN
  SELECT * INTO m FROM public.maladies_parametres WHERE cle = p_cle;
  IF NOT FOUND OR NOT m.actif THEN RETURN NULL; END IF;
  IF NOT EXISTS (SELECT 1 FROM public.personnages_donnees WHERE name = p_nom) THEN RETURN NULL; END IF;
  PERFORM public.maladies_echeances_interne(p_nom);
  IF EXISTS (SELECT 1 FROM public.personnages_maladies WHERE nom = p_nom AND cle = p_cle AND guerie_le IS NULL) THEN RETURN NULL; END IF;
  INSERT INTO public.personnages_maladies (nom, cle, source, fin)
  VALUES (p_nom, p_cle, coalesce(p_source, 'inconnue'), now() + make_interval(hours => m.duree_heures)) RETURNING id INTO v_id;
  PERFORM public.maladie_effets_synchroniser_interne(p_nom);
  RETURN jsonb_build_object('id', v_id, 'cle', p_cle, 'nom_mj', m.nom_mj, 'symptome_mj', m.symptome_mj, 'duree_heures', m.duree_heures);
END $function$;

CREATE OR REPLACE FUNCTION public.maladies_nuit_interne(p_nom text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE v jsonb;
BEGIN
  WITH g AS (
    UPDATE public.personnages_maladies SET guerie_le = now(),
           motif_fin = CASE WHEN traitee_le IS NOT NULL THEN 'medicament' ELSE 'echeance' END
     WHERE nom = p_nom AND guerie_le IS NULL AND (traitee_le IS NOT NULL OR fin <= now())
    RETURNING cle)
  SELECT coalesce(jsonb_agg(cle), '[]'::jsonb) INTO v FROM g;
  IF jsonb_array_length(v) > 0 THEN PERFORM public.maladie_effets_synchroniser_interne(p_nom); END IF;
  RETURN v;
END $function$;

CREATE OR REPLACE FUNCTION public.achat_requete_ouvrir_interne(p_acteur text, p_requete uuid, p_type text, p_params jsonb)
 RETURNS jsonb LANGUAGE sql AS $$ SELECT NULL::jsonb $$;
CREATE OR REPLACE FUNCTION public.achat_requete_clore_interne(p_acteur text, p_requete uuid, p_type text, p_params jsonb, p_rep jsonb)
 RETURNS jsonb LANGUAGE sql AS $$ SELECT p_rep $$;
CREATE OR REPLACE FUNCTION public.sovarka_inventaire_ligne_interne(p_inv jsonb, p_cles text[])
 RETURNS jsonb LANGUAGE sql AS $$
  SELECT coalesce((SELECT jsonb_build_object('i', o.ord - 1, 'cle', o.e->>'stackKey') FROM jsonb_array_elements(coalesce(p_inv, '[]')) WITH ORDINALITY o(e, ord)
                    WHERE o.e->>'stackKey' = ANY (p_cles) ORDER BY o.ord LIMIT 1), '{"i":-1}'::jsonb) $$;
CREATE OR REPLACE FUNCTION public.sovarka_inventaire_retirer_interne(p_nom text, p_ligne jsonb, p_qte integer)
 RETURNS void LANGUAGE sql AS $$
  UPDATE public.personnages_donnees SET inventory = CASE
      WHEN coalesce((inventory->((p_ligne->>'i')::integer)->>'qty')::integer, 1) > p_qte
        THEN jsonb_set(inventory, ARRAY[(p_ligne->>'i'), 'qty'], to_jsonb((inventory->((p_ligne->>'i')::integer)->>'qty')::integer - p_qte))
      ELSE public.inventaire_retirer_position(inventory, (p_ligne->>'i')::integer) END
   WHERE name = p_nom $$;

CREATE OR REPLACE FUNCTION public.medicament_prendre(p_requete uuid, p_medicament text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE a text := public.mon_personnage(); prm jsonb := jsonb_build_object('medicament', p_medicament); v jsonb; l jsonb;
        v_cle text; v_inv jsonb; i integer;
BEGIN
  IF a IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'acteur_non_authentifie'); END IF;
  v := public.achat_requete_ouvrir_interne(a, p_requete, 'medicament_prendre', prm);
  IF v IS NOT NULL THEN RETURN v; END IF;
  SELECT cle INTO v_cle FROM public.maladies_parametres WHERE medicament = p_medicament;
  IF v_cle IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'medicament_inconnu'); END IF;
  SELECT coalesce(inventory, '[]'::jsonb) INTO v_inv FROM public.personnages_donnees WHERE name = a FOR UPDATE;
  PERFORM public.maladies_echeances_interne(a);
  IF NOT EXISTS (SELECT 1 FROM public.personnages_maladies WHERE nom = a AND cle = v_cle AND guerie_le IS NULL) THEN
    RETURN jsonb_build_object('ok', false, 'raison', 'aucune_maladie_a_traiter'); END IF;
  IF EXISTS (SELECT 1 FROM public.personnages_maladies WHERE nom = a AND cle = v_cle AND guerie_le IS NULL AND traitee_le IS NOT NULL) THEN
    RETURN jsonb_build_object('ok', false, 'raison', 'deja_traitee'); END IF;
  l := public.sovarka_inventaire_ligne_interne(v_inv, ARRAY['medicament_' || p_medicament]);
  IF (l->>'i')::integer < 0 THEN RETURN jsonb_build_object('ok', false, 'raison', 'pas_de_medicament'); END IF;
  PERFORM public.sovarka_inventaire_retirer_interne(a, l, 1);
  UPDATE public.personnages_maladies SET traitee_le = now() WHERE nom = a AND cle = v_cle AND guerie_le IS NULL;
  v := jsonb_build_object('ok', true, 'medicament', p_medicament, 'maladie', v_cle)
    || (SELECT jsonb_build_object('inventory', inventory, 'revision', inventory_revision) FROM public.personnages_donnees WHERE name = a);
  RETURN public.achat_requete_clore_interne(a, p_requete, 'medicament_prendre', prm, v);
END $function$;

CREATE OR REPLACE FUNCTION public.admin_maladie_modifier(p_cle text, p_parametres jsonb)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE k text; v_admin text := auth.uid()::text;
BEGIN
  PERFORM public.admin_exiger();
  IF NOT EXISTS (SELECT 1 FROM public.maladies_parametres WHERE cle = p_cle) THEN
    RETURN jsonb_build_object('ok', false, 'raison', 'maladie_inconnue'); END IF;
  IF jsonb_typeof(p_parametres) IS DISTINCT FROM 'object' THEN RETURN jsonb_build_object('ok', false, 'raison', 'parametres_invalides'); END IF;
  FOR k IN SELECT jsonb_object_keys(p_parametres) LOOP
    IF k NOT IN ('actif','deltas','pa_max','sans_bonus_arme','duree_heures','medicament','contagieuse','risques',
                 'prix_fixe','stock_journalier','nom_mj','symptome_mj') THEN
      RETURN jsonb_build_object('ok', false, 'raison', 'parametre_inconnu', 'parametre', k); END IF;
  END LOOP;
  UPDATE public.maladies_parametres m SET
    actif           = CASE WHEN p_parametres ? 'actif' THEN (p_parametres->>'actif')::boolean ELSE m.actif END,
    deltas          = CASE WHEN p_parametres ? 'deltas' THEN p_parametres->'deltas' ELSE m.deltas END,
    pa_max          = CASE WHEN p_parametres ? 'pa_max' THEN (p_parametres->>'pa_max')::integer ELSE m.pa_max END,
    sans_bonus_arme = CASE WHEN p_parametres ? 'sans_bonus_arme' THEN (p_parametres->>'sans_bonus_arme')::boolean ELSE m.sans_bonus_arme END,
    duree_heures    = CASE WHEN p_parametres ? 'duree_heures' THEN (p_parametres->>'duree_heures')::integer ELSE m.duree_heures END,
    medicament      = CASE WHEN p_parametres ? 'medicament' THEN p_parametres->>'medicament' ELSE m.medicament END,
    contagieuse     = CASE WHEN p_parametres ? 'contagieuse' THEN (p_parametres->>'contagieuse')::boolean ELSE m.contagieuse END,
    risques         = CASE WHEN p_parametres ? 'risques' THEN p_parametres->'risques' ELSE m.risques END,
    prix_fixe       = CASE WHEN p_parametres ? 'prix_fixe' THEN (p_parametres->>'prix_fixe')::integer ELSE m.prix_fixe END,
    stock_journalier= CASE WHEN p_parametres ? 'stock_journalier' THEN (p_parametres->>'stock_journalier')::integer ELSE m.stock_journalier END,
    nom_mj          = CASE WHEN p_parametres ? 'nom_mj' THEN nullif(btrim(p_parametres->>'nom_mj'), '') ELSE m.nom_mj END,
    symptome_mj     = CASE WHEN p_parametres ? 'symptome_mj' THEN nullif(btrim(p_parametres->>'symptome_mj'), '') ELSE m.symptome_mj END,
    updated_at = now(), updated_by = v_admin
  WHERE m.cle = p_cle;
  INSERT INTO public.admin_journal (admin_user_id, admin_libelle, action, parametres, ok)
  VALUES (auth.uid(), 'MJ', 'maladie_modifier', jsonb_build_object('cle', p_cle) || p_parametres, true);
  RETURN jsonb_build_object('ok', true, 'parametres', (SELECT to_jsonb(m) FROM public.maladies_parametres m WHERE m.cle = p_cle));
END $function$;

CREATE OR REPLACE FUNCTION public.escort_intimite(p_requete uuid, p_escort text, p_palier_vise integer DEFAULT NULL::integer)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE moi text := public.mon_personnage(); v_ancien jsonb; v_pnj text; v_avant integer; v_palier integer;
  v_cout_etape integer := 0; c_prestation constant integer := 300; v_total integer; v_debit jsonb;
  v_moral integer; v_hp integer; v_ent integer; v_ent_gagne integer; v_rep jsonb;
BEGIN
  IF moi IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'acteur_non_authentifie'); END IF;
  IF public.mineur_refus_interne(moi, NULL, 'escort') IS NOT NULL THEN /* mineurs_v1 */
    RETURN public.mineur_refus_interne(moi, NULL, 'escort'); END IF;
  IF p_requete IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'requete_requise'); END IF;
  PERFORM pg_advisory_xact_lock(hashtextextended('escort_intimite:' || moi, 0));
  SELECT reponse INTO v_ancien FROM public.escorts_intimite_recus WHERE requete = p_requete AND acteur = moi;
  IF v_ancien IS NOT NULL THEN RETURN v_ancien || jsonb_build_object('rejeu', true); END IF;
  v_pnj := public.escort_engagee_interne(moi, p_escort);
  IF v_pnj IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'escort_non_engagee'); END IF;
  PERFORM 1 FROM public.pnj_membres WHERE id = v_pnj FOR UPDATE;
  v_avant := public.escort_palier_interne(v_pnj, moi);
  v_palier := v_avant;
  IF p_palier_vise IS NOT NULL THEN
    IF p_palier_vise <> v_avant + 1 OR p_palier_vise > 3 THEN
      RETURN jsonb_build_object('ok', false, 'raison', 'etape_invalide', 'palier', v_avant); END IF;
    v_cout_etape := CASE p_palier_vise WHEN 1 THEN 100 WHEN 2 THEN 300 WHEN 3 THEN 600 END;
    v_palier := p_palier_vise;
  END IF;
  v_total := v_cout_etape + c_prestation;
  v_debit := public.debiter_fonds_ordinaires(moi, v_total);
  IF coalesce((v_debit->>'ok')::boolean, false) IS NOT TRUE THEN
    RETURN jsonb_build_object('ok', false, 'raison', coalesce(v_debit->>'raison', 'fonds_insuffisants'), 'total', v_total, 'palier', v_avant); END IF;
  INSERT INTO public.escorts_complicite (pnj_id, employeur, palier, updated_at) VALUES (v_pnj, moi, v_palier, now())
  ON CONFLICT (pnj_id) DO UPDATE SET employeur = EXCLUDED.employeur, palier = EXCLUDED.palier, updated_at = now();
  v_moral := (ARRAY[15, 18, 22, 28])[v_palier + 1];
  v_hp    := (ARRAY[5, 6, 7, 10])[v_palier + 1];
  v_ent   := (ARRAY[2, 2, 3, 4])[v_palier + 1];
  UPDATE public.personnages_donnees
     SET moral = least(100, coalesce(moral, 75) + v_moral), hp = least(100, coalesce(hp, 100) + v_hp), updated_at = now()
   WHERE name = moi;
  v_ent_gagne := public.ent_gain_quotidien_interne(moi, v_ent);
  SELECT jsonb_build_object('ok', true, 'escort', p_escort, 'palier_avant', v_avant, 'palier', v_palier,
           'cout_etape', v_cout_etape, 'cout_prestation', c_prestation, 'total', v_total,
           'gain_moral', v_moral, 'gain_hp', v_hp, 'gain_ent', v_ent_gagne,
           'moral', d.moral, 'hp', d.hp, 'ent', coalesce((d.stats->>'ENT')::numeric, 0),
           'liquide', d.liquide, 'arg', d.arg,
           'solde_national', coalesce((SELECT solde FROM public.comptes_bancaires WHERE personnage = moi AND banque = 'nationale'), 0))
    INTO v_rep FROM public.personnages_donnees d WHERE d.name = moi;
  INSERT INTO public.escorts_intimite_recus (requete, acteur, reponse) VALUES (p_requete, moi, v_rep);
  RETURN v_rep;
END $function$;

CREATE OR REPLACE FUNCTION public.pharmacie_point_interne(p_acteur text)
 RETURNS pharmacies_points_vente
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
  SELECT pv.* FROM public.pharmacies_points_vente pv JOIN public.personnages_donnees d ON d.name = p_acteur
   WHERE d.country = pv.pays AND d.current_city = pv.ville AND d.current_building = pv.batiment AND d.current_room = pv.piece;
$function$;
CREATE OR REPLACE FUNCTION public.pharmacie_prix_interne(p_cle_maladie text)
 RETURNS integer
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
  SELECT coalesce(m.prix_fixe, (SELECT greatest(1, round(2 * prix_base))::integer FROM public.ressources_economie WHERE cle = 'medicaments'), 44)
    FROM public.maladies_parametres m WHERE m.cle = p_cle_maladie;
$function$;
CREATE OR REPLACE FUNCTION public.pharmacie_prix_pays_interne(p_cle_maladie text, p_pays text)
 RETURNS integer
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
  SELECT CASE WHEN p_cle_maladie = 'coup_de_chaleur' AND p_pays = 'soviet'
              THEN greatest(1, round(public.pharmacie_prix_interne(p_cle_maladie) / 2.0))::integer
              ELSE public.pharmacie_prix_interne(p_cle_maladie) END;
$function$;
CREATE OR REPLACE FUNCTION public.pharmacie_stock_point_interne(p_point text, p_cle_maladie text)
 RETURNS integer
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
  WITH s AS (SELECT stock_journalier::numeric AS n FROM public.maladies_parametres WHERE cle = p_cle_maladie),
       g AS (SELECT pv.id, floor(s.n * pv.part) AS base, s.n * pv.part - floor(s.n * pv.part) AS reste
               FROM public.pharmacies_points_vente pv, s
              WHERE pv.pays = (SELECT pays FROM public.pharmacies_points_vente WHERE id = p_point) AND pv.part > 0),
       t AS (SELECT round(sum(pv.part) * (SELECT n FROM s)) AS total FROM public.pharmacies_points_vente pv
              WHERE pv.pays = (SELECT pays FROM public.pharmacies_points_vente WHERE id = p_point) AND pv.part > 0),
       r AS (SELECT id, base, row_number() OVER (ORDER BY reste DESC, id) AS rang FROM g)
  SELECT coalesce((SELECT (base + CASE WHEN rang <= (SELECT total FROM t) - (SELECT sum(base) FROM g) THEN 1 ELSE 0 END)::integer
                     FROM r WHERE id = p_point), 0);
$function$;

CREATE OR REPLACE FUNCTION public.point_eau_etat()
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE a text := public.mon_personnage(); d public.personnages_donnees; pe public.climat_points_eau;
        v_max integer := public.aliments_param_interne('point_eau_bouteilles_tranche', 10)::integer;
        v_tr timestamptz := public.point_eau_tranche_interne(); v_deja integer; v_place integer;
BEGIN
  IF a IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'acteur_non_authentifie'); END IF;
  SELECT * INTO d FROM public.personnages_donnees WHERE name = a;
  SELECT * INTO pe FROM public.climat_points_eau p WHERE p.actif AND p.pays = d.country AND p.ville = d.current_city
     AND p.noeud = d.current_street AND d.current_building IS NULL ORDER BY p.id LIMIT 1;
  IF pe.id IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'pas_de_point_eau'); END IF;
  SELECT coalesce(sum(quantite), 0) INTO v_deja FROM public.climat_points_eau_puisages WHERE nom = a AND tranche = v_tr;
  v_place := public.inventaire_place_restante(d.inventory);
  RETURN jsonb_build_object('ok', true, 'point', jsonb_build_object('id', pe.id, 'nom', pe.nom, 'image', pe.image),
    'max_tranche', v_max, 'deja', v_deja, 'restant', greatest(0, v_max - v_deja), 'place', v_place,
    'pa', coalesce(d.pa, 0), 'cout_pa', 1,
    'fin_tranche', v_tr + interval '6 hours',
    'heure_fin', extract(hour FROM (v_tr + interval '6 hours') AT TIME ZONE 'Europe/Paris')::integer);
END $function$;

CREATE OR REPLACE FUNCTION public.point_eau_puiser(p_requete uuid, p_quantite integer)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE a text := public.mon_personnage(); prm jsonb; v jsonb; e jsonb; p jsonb; d public.personnages_donnees;
        v_n integer; v_tr timestamptz := public.point_eau_tranche_interne();
BEGIN
  IF a IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'acteur_non_authentifie'); END IF;
  IF p_requete IS NULL THEN RETURN jsonb_build_object('ok', false, 'raison', 'requete_absente'); END IF;
  IF p_quantite IS NULL OR p_quantite < 1 THEN RETURN jsonb_build_object('ok', false, 'raison', 'quantite_invalide'); END IF;
  prm := jsonb_build_object('quantite', p_quantite);
  v := public.achat_requete_ouvrir_interne(a, p_requete, 'point_eau_puiser', prm);
  IF v IS NOT NULL THEN RETURN v; END IF;
  PERFORM pg_advisory_xact_lock(hashtextextended('point_eau:' || a, 0));
  SELECT * INTO d FROM public.personnages_donnees WHERE name = a FOR UPDATE;
  e := public.point_eau_etat();
  IF coalesce((e->>'ok')::boolean, false) IS NOT TRUE THEN RETURN e; END IF;
  IF (e->>'restant')::integer <= 0 THEN RETURN e || jsonb_build_object('ok', false, 'raison', 'plafond_atteint'); END IF;
  IF (e->>'place')::integer <= 0 THEN RETURN e || jsonb_build_object('ok', false, 'raison', 'inventaire_plein'); END IF;
  v_n := least(p_quantite, (e->>'restant')::integer, (e->>'place')::integer);
  p := public.payer_ordre_interne(a, 'puiser_eau', 1, 0);
  IF coalesce((p->>'ok')::boolean, false) IS NOT TRUE THEN RETURN p; END IF;
  v := public.inventaire_deposer_interne(a, public.climat_objet_interne('bouteille_eau', v_n));
  IF coalesce((v->>'ok')::boolean, false) IS NOT TRUE THEN RAISE EXCEPTION 'point_eau_depot_refuse:%', v->>'raison'; END IF;
  INSERT INTO public.climat_points_eau_puisages (nom, tranche, quantite) VALUES (a, v_tr, v_n)
    ON CONFLICT (nom, tranche) DO UPDATE SET quantite = climat_points_eau_puisages.quantite + EXCLUDED.quantite;
  RETURN public.achat_requete_clore_interne(a, p_requete, 'point_eau_puiser', prm,
    v || jsonb_build_object('ok', true, 'quantite', v_n, 'pa_depenses', 1,
      'restant', (e->>'restant')::integer - v_n, 'heure_fin', (e->>'heure_fin')::integer,
      'pa', (SELECT pa FROM public.personnages_donnees WHERE name = a)));
END $function$;
