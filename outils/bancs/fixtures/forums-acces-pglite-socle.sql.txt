CREATE ROLE authenticated;
CREATE ROLE anon;
CREATE TABLE personnages_donnees (name text PRIMARY KEY, country text, current_city text, nationalite text, domicile jsonb, career text);
CREATE TABLE organisations (id text PRIMARY KEY, data text);
CREATE TABLE forum_topics (id text PRIMARY KEY, forum_id text, title text, author text, country text, time text, views int, replies int,
  last_post_author text, last_post_time text, author_is_org boolean, author_secret boolean, author_real text, author_org_id text, author_org_icon text);
CREATE TABLE forum_posts (id text PRIMARY KEY, topic_id text, author text, content text, time text, author_is_org boolean, author_secret boolean,
  content_blocks jsonb, content_layout jsonb, author_real text, author_org_id text, author_org_icon text);
CREATE FUNCTION mon_personnage() RETURNS text LANGUAGE sql STABLE AS $$ SELECT nullif(current_setting('rp.moi', true), '') $$;
CREATE FUNCTION poste_exerce(p_nom text) RETURNS text LANGUAGE sql STABLE AS $$ SELECT NULL::text $$;
CREATE FUNCTION personnage_nationalite(p_nom text) RETURNS text LANGUAGE sql STABLE SECURITY DEFINER SET search_path TO 'public', 'pg_temp' AS $$
  SELECT coalesce(nationalite, country) FROM public.personnages_donnees WHERE name = p_nom
$$;

CREATE OR REPLACE FUNCTION public.forum_lecture_autorisee(p_forum text, p_pays text)
 RETURNS boolean
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE
  v_nom text := public.mon_personnage();
  v_pays text; v_ville text; v_nat text; v_dom jsonb; v_local text;
BEGIN
  IF p_forum IS NULL THEN RETURN false; END IF;
  IF left(p_forum,4) = 'org_' THEN
    IF v_nom IS NULL THEN RETURN false; END IF;
    RETURN EXISTS (
      SELECT 1 FROM public.organisations o
       WHERE o.id = substring(p_forum from 5)
         AND CASE WHEN o.data IS JSON OBJECT THEN (
           (o.data::jsonb ->> 'chef') = v_nom
           OR EXISTS (
             SELECT 1 FROM jsonb_array_elements(
               CASE WHEN jsonb_typeof(o.data::jsonb -> 'membres') = 'array'
                 THEN o.data::jsonb -> 'membres' ELSE '[]'::jsonb END
             ) m(elem)
             WHERE coalesce(m.elem ->> 'nom', m.elem #>> '{}') = v_nom
           )
         ) ELSE false END
    );
  END IF;
  IF p_forum = 'national' OR left(p_forum,6) = 'local_' THEN
    IF v_nom IS NULL THEN RETURN false; END IF;
    SELECT country, current_city, coalesce(nationalite, country), domicile INTO v_pays, v_ville, v_nat, v_dom
      FROM public.personnages_donnees WHERE name = v_nom;
    v_local := substring(p_forum from 7);
    IF p_pays IS NOT DISTINCT FROM v_nat THEN
      IF p_forum = 'national' THEN RETURN true; END IF;
      RETURN (v_pays IS NOT DISTINCT FROM v_nat AND v_ville IS NOT DISTINCT FROM v_local)
          OR (v_dom ->> 'country' IS NOT DISTINCT FROM v_nat AND v_dom ->> 'city' IS NOT DISTINCT FROM v_local);
    ELSIF p_pays IS NOT DISTINCT FROM v_pays THEN
      IF p_forum = 'national' THEN RETURN true; END IF;
      RETURN v_ville IS NOT DISTINCT FROM v_local;
    END IF;
    RETURN false;
  END IF;
  RETURN true;
END;
$function$;

CREATE OR REPLACE FUNCTION public.forum_ecriture_autorisee(p_forum text, p_pays text)
 RETURNS boolean
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
BEGIN
  IF NOT public.forum_lecture_autorisee(p_forum, p_pays) THEN RETURN false; END IF;
  IF p_forum = 'national' AND p_pays IS DISTINCT FROM public.personnage_nationalite(public.mon_personnage()) THEN
    RETURN false;
  END IF;
  RETURN true;
END;
$function$;

CREATE OR REPLACE FUNCTION public.forum_publication_jauges_interne(p_acteur text, p_requete uuid, p_message jsonb)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE a public.personnages_donnees; t public.forum_topics; o jsonb;
 f text; auteur text; orga text; nom_reel text; sujet text; post_id text; temps text;
 nouveau boolean; cree text; secret boolean; icone text; membre boolean;
BEGIN
 IF p_acteur IS DISTINCT FROM public.mon_personnage() OR p_requete IS NULL THEN
  RETURN jsonb_build_object('ok',false,'raison','acteur_non_authentifie');END IF;
 IF jsonb_typeof(p_message) IS DISTINCT FROM 'object' OR
    p_message-ARRAY['forum','topic','titre','contenu','blocs','mise_en_page','organisation','secret','icone','pays']<>'{}'::jsonb THEN
  RETURN jsonb_build_object('ok',false,'raison','champs_interdits');END IF;
 SELECT * INTO a FROM public.personnages_donnees WHERE name=p_acteur FOR UPDATE;
 IF nullif(btrim(p_message->>'contenu'),'') IS NULL OR
    jsonb_typeof(coalesce(p_message->'blocs','[]'::jsonb))<>'array' THEN
  RETURN jsonb_build_object('ok',false,'raison','publication_vide');END IF;
 nouveau:=nullif(p_message->>'topic','') IS NULL;
 IF nouveau THEN
  f:=p_message->>'forum';sujet:='topic-'||p_requete::text;
  IF nullif(btrim(p_message->>'titre'),'')IS NULL THEN RETURN jsonb_build_object('ok',false,'raison','titre_vide');END IF;
 ELSE
  SELECT * INTO t FROM public.forum_topics WHERE id=p_message->>'topic' FOR UPDATE;
  IF NOT FOUND THEN RETURN jsonb_build_object('ok',false,'raison','sujet_introuvable');END IF;
  f:=t.forum_id;sujet:=t.id;
  IF f NOT IN('international','sport')AND left(f,4)<>'org_'AND t.country IS DISTINCT FROM a.country AND t.country IS DISTINCT FROM coalesce(a.nationalite, a.country) THEN
   RETURN jsonb_build_object('ok',false,'raison','territoire_invalide');END IF;
 END IF;
 -- Les mêmes restrictions que FORUMS_BASE/getForums/canAccessForum et les éditeurs.
 -- Presse et Présidence : lecture et réponses publiques, création réservée.
 IF f IS NULL THEN RETURN jsonb_build_object('ok',false,'raison','forum_inconnu');END IF;
 IF NOT public.forum_ecriture_autorisee(f,CASE WHEN nouveau THEN CASE WHEN f='national' OR left(f,6)='local_' THEN coalesce(nullif(p_message->>'pays',''), a.country) ELSE a.country END ELSE t.country END)THEN
  RETURN jsonb_build_object('ok',false,'raison','forum_inaccessible');END IF;
 IF f='presidence'AND nouveau AND public.poste_exerce(a.name)/*pp_forum*/ IS DISTINCT FROM 'president' THEN
  RETURN jsonb_build_object('ok',false,'raison','poste_requis');END IF;
 IF f='presse'AND nouveau AND a.career IS DISTINCT FROM 'press' THEN
  RETURN jsonb_build_object('ok',false,'raison','carriere_requise');END IF;
 IF f='gouvernement'AND NOT(coalesce(public.poste_exerce(a.name)/*pp_forum*/,'')=ANY(ARRAY['president','pm','min_int','min_fin','min_just','min_def','min_info','min_ae']))THEN
  RETURN jsonb_build_object('ok',false,'raison','poste_requis');END IF;
 IF left(f,4)='org_' THEN
  SELECT CASE WHEN data IS JSON OBJECT THEN data::jsonb ELSE '{}'::jsonb END INTO o FROM public.organisations WHERE id=substr(f,5);
  membre:=public.forum_lecture_autorisee(f,a.country);
  IF NOT membre THEN RETURN jsonb_build_object('ok',false,'raison','membre_requis');END IF;
 ELSIF f NOT IN('national','presidence','assemblee','gouvernement','presse','international','sport',
               'local_'||a.current_city,'local_'||coalesce(a.domicile->>'city',''),'tribunal_'||a.current_city)THEN
  RETURN jsonb_build_object('ok',false,'raison','forum_inconnu');END IF;
 auteur:=a.name;orga:=nullif(p_message->>'organisation','');
 IF orga IS NOT NULL THEN
  SELECT CASE WHEN data IS JSON OBJECT THEN data::jsonb ELSE '{}'::jsonb END INTO o FROM public.organisations WHERE id=orga;
  IF o->>'chef' IS DISTINCT FROM a.name THEN RETURN jsonb_build_object('ok',false,'raison','chef_organisation_requis');END IF;
  IF coalesce((o->>'dissoute')::boolean,false)THEN RETURN jsonb_build_object('ok',false,'raison','organisation_dissoute');END IF;
  auteur:=o->>'nom';nom_reel:=a.name;
  IF nullif(auteur,'')IS NULL THEN RETURN jsonb_build_object('ok',false,'raison','organisation_invalide');END IF;
 END IF;
 secret:=orga IS NOT NULL AND NOT coalesce((o->>'visible')::boolean,true);icone:=CASE WHEN orga IS NOT NULL THEN p_message->>'icone' END;
 temps:=to_char(now()AT TIME ZONE'Europe/Paris','DD/MM/YYYY HH24:MI');post_id:='post-'||p_requete::text;
 IF nouveau THEN
  INSERT INTO public.forum_topics(id,forum_id,title,author,country,time,views,replies,last_post_author,last_post_time,
    author_is_org,author_secret,author_real,author_org_id,author_org_icon)
  VALUES(sujet,f,btrim(p_message->>'titre'),auteur,CASE WHEN f='national' OR left(f,6)='local_' THEN coalesce(nullif(p_message->>'pays',''), a.country) ELSE a.country END,temps,1,0,auteur,temps,orga IS NOT NULL,secret,nom_reel,orga,icone)RETURNING id INTO cree;
  IF cree IS NULL THEN RAISE EXCEPTION 'forum_ecriture_refusee' USING ERRCODE='42501';END IF;
 END IF;
 cree:=NULL;
 INSERT INTO public.forum_posts(id,topic_id,author,content,time,author_is_org,author_secret,content_blocks,content_layout,author_real,author_org_id,author_org_icon)
 VALUES(post_id,sujet,auteur,p_message->>'contenu',temps,orga IS NOT NULL,secret,coalesce(p_message->'blocs','[]'::jsonb),p_message->'mise_en_page',nom_reel,orga,icone)RETURNING id INTO cree;
 IF cree IS NULL THEN RAISE EXCEPTION 'forum_ecriture_refusee' USING ERRCODE='42501';END IF;
 UPDATE public.forum_topics SET replies=(SELECT greatest(0,count(*)-1)FROM public.forum_posts WHERE topic_id=sujet),last_post_author=auteur,last_post_time=temps WHERE id=sujet;
 RETURN jsonb_build_object('ok',true,'topic',sujet,'post',post_id,'temps',temps,'gain_pop',CASE WHEN nouveau THEN 2 ELSE 1 END);
EXCEPTION WHEN insufficient_privilege OR check_violation OR foreign_key_violation OR unique_violation THEN
 -- Le bloc entier est annulé : aucun sujet orphelin, message sans bonus, ni bonus sans message.
 RETURN jsonb_build_object('ok',false,'raison','publication_refusee');
END;$function$;
