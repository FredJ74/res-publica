CREATE OR REPLACE FUNCTION public.approvisionner_chantier(a text, p text, v text, e text, b jsonb, s jsonb, t numeric) RETURNS jsonb LANGUAGE plpgsql AS $$
DECLARE mm text; manque numeric; qte numeric; dep numeric := 0; achats jsonb := '{}'; st jsonb := '{}';
BEGIN
  FOREACH mm IN ARRAY ARRAY['bois','minerai','metal'] LOOP
    st := st || jsonb_build_object(mm, greatest(0, coalesce((s->>mm)::numeric, 0)));
    manque := greatest(0, coalesce((b->>mm)::numeric, 0) - (st->>mm)::numeric);
    qte := least(manque, (SELECT q FROM public.entrepot_test WHERE entrepot_test.m = mm), floor(greatest(0, t - dep)));
    CONTINUE WHEN qte <= 0;
    UPDATE public.entrepot_test SET q = q - qte WHERE entrepot_test.m = mm;
    achats := achats || jsonb_build_object(mm, qte); dep := dep + qte;
    st := st || jsonb_build_object(mm, (st->>mm)::numeric + qte);
  END LOOP;
  RETURN jsonb_build_object('ok', true, 'achats', achats, 'depense', dep, 'stockChantier', st);
END $$;
